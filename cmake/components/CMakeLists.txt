# Registered owned paths; owners add their own CMakeLists.txt when claimed.
# Explicit opt-in prevents unrelated/incomplete shared-checkout work being built.
function(prism_add_component_check id)
  if(BUILD_TESTING)
    string(TOLOWER "${id}" lower_id)
    add_custom_target(check-${lower_id}
      COMMAND ${CMAKE_CTEST_COMMAND} --output-on-failure --no-tests=error -L "^${id}$"
      WORKING_DIRECTORY "${PROJECT_BINARY_DIR}" DEPENDS ${ARGN})
  endif()
endfunction()
function(prism_component id)
  option(PRISM_BUILD_${id} "Build registered ${id} components" OFF)
  if(NOT PRISM_BUILD_${id})
    return()
  endif()
  set(found FALSE)
  foreach(path IN LISTS ARGN)
    if(EXISTS "${PROJECT_SOURCE_DIR}/${path}/CMakeLists.txt")
      add_subdirectory("${PROJECT_SOURCE_DIR}/${path}" "${PROJECT_BINARY_DIR}/components/${path}")
      set(found TRUE)
    endif()
  endforeach()
  if(NOT found)
    message(FATAL_ERROR "${id} requested but no owned component CMakeLists.txt exists yet")
  endif()
endfunction()
